<script>
(function(){
  const nav = document.querySelector('.insTabs');
  if(!nav) return;

  const links = Array.from(nav.querySelectorAll('.insTabs__link'));
  const getOffset = () => {
    const h = nav.getBoundingClientRect().height || 58;
    return h + 12; // ✅ small breathing space so title is visible
  };

  // Smooth scroll with correct offset so sections are NOT hidden under sticky tabs
  links.forEach(a=>{
    a.addEventListener('click', (e)=>{
      const href = a.getAttribute('href');
      if(!href || !href.startsWith('#')) return;

      const el = document.querySelector(href);
      if(!el) return;

      e.preventDefault();
      const y = el.getBoundingClientRect().top + window.pageYOffset - getOffset();
      window.scrollTo({ top:y, behavior:'smooth' });
      history.replaceState(null, '', href);
    });
  });

  // Active state on scroll
  const targets = links
    .map(a => document.querySelector(a.getAttribute('href')))
    .filter(Boolean);

  const setActive = (id) => {
    links.forEach(a=>{
      a.classList.toggle('is-active', a.getAttribute('href') === '#'+id);
    });
  };

  const isElementor =
    document.body.classList.contains('elementor-editor-active') ||
    document.body.classList.contains('elementor-editor-preview') ||
    !!document.querySelector('#elementor-editor-wrapper');

  if(isElementor) return;
  if(!('IntersectionObserver' in window)) return;

  const io = new IntersectionObserver((entries)=>{
    // pick the entry closest to top that is visible
    const visible = entries
      .filter(e=>e.isIntersecting)
      .sort((a,b)=>b.intersectionRatio-a.intersectionRatio)[0];

    if(visible && visible.target && visible.target.id){
      setActive(visible.target.id);
    }
  }, { root:null, threshold:[0.2,0.35,0.5,0.65] });

  targets.forEach(t=>{
    if(t.id) io.observe(t);
  });
})();
</script>

(function(){

      ],
      chips:["Supply chain","Capacity","Throughput"]
    },
    c3:{
      title:"Cost, Productivity & Margin Stress",
      bullets:[
        "Rising material and overhead costs across long program lifecycles",
        "Limited cost transparency at program and platform level",
        "Inefficiencies driven by legacy processes and manual workflows",
        "Pressure to improve competitiveness without compromising quality"
      ],
      chips:["Cost transparency","Productivity","Margins"]
    },
    c4:{
      title:"Quality, Compliance & Risk Exposure",
      bullets:[
        "Stringent certification, audit, and regulatory requirements",
        "Increasing scrutiny on traceability, documentation, and data integrity",
        "Program delays due to quality escapes or compliance gaps",
        "Limited enterprise-wide risk visibility"
      ],
      chips:["Certification","Traceability","Enterprise risk"]
    },
    c5:{
      title:"Digital & Technology Gaps",
      bullets:[
        "Fragmented systems across engineering, manufacturing, and supply chain",
        "Low adoption of digital thread, digital twin, and advanced analytics",
        "Limited use of data for program-level decision-making",
        "Challenges in scaling Industry 4.0 initiatives across plants and partners"
      ],
      chips:["Digital thread","Analytics","Industry 4.0"]
    }
  };

  const root = document.querySelector('#ad-challenges');
  if(!root) return;

  function applyMobileClass(){
    const mqMobile = window.matchMedia && window.matchMedia('(max-width: 820px)').matches;

    // Elementor responsive preview helpers
    const elMobile =
      document.body.classList.contains('elementor-device-mobile') ||
      document.body.classList.contains('e--ua-device-mobile');

    const mobile = mqMobile || elMobile;

    root.classList.toggle('is-mobile', mobile);
    return mobile;
  }
  applyMobileClass();
  window.addEventListener('resize', applyMobileClass);

  const tabs = Array.from(root.querySelectorAll('.adK__item'));
  const fade = root.querySelector('#adK-fade');
  const title = root.querySelector('#adK-title');
  const bullets = root.querySelector('#adK-bullets');
  const chips = root.querySelector('#adK-chips');

  function isMobile(){
    return (window.matchMedia && window.matchMedia('(max-width: 820px)').matches) || root.classList.contains('is-mobile');
  }

  function renderDesktop(key){
    const d = data[key];
    if(!d || !fade) return;

    fade.classList.add('is-switching');
    setTimeout(()=>{
      title.textContent = d.title;

      bullets.innerHTML = d.bullets.map(x =>
        `<li><span class="adK__dot"></span><span>${x}</span></li>`
      ).join('');

      chips.innerHTML = d.chips.map(x =>
        `<span class="adK__chip"><i></i>${x}</span>`
      ).join('');

      fade.classList.remove('is-switching');
    }, 160);
  }

  function renderAccordionItem(btn){
    const key = btn.dataset.key;
    const d = data[key];
    if(!d) return;

    const acc = btn.querySelector('.adK__acc');
    if(!acc) return;

    acc.innerHTML = `
      <ul class="adK__accBullets">
        ${d.bullets.map(x => `<li><span class="adK__dot"></span><span>${x}</span></li>`).join('')}
      </ul>
      <div class="adK__accChips">
        ${d.chips.map(x => `<span class="adK__chip"><i></i>${x}</span>`).join('')}
      </div>
    `;
  }

  tabs.forEach(btn => renderAccordionItem(btn));

  function setActive(btn){
    tabs.forEach(x=>{
      x.setAttribute('aria-selected','false');
      x.setAttribute('aria-expanded','false');
    });
    btn.setAttribute('aria-selected','true');

    if(isMobile()){
      btn.setAttribute('aria-expanded','true');
    }else{
      btn.setAttribute('aria-expanded','false');
      renderDesktop(btn.dataset.key);
    }
  }

  // Desktop default
  renderDesktop('c1');

  // Mobile: open first card by default
  if(isMobile()){
    tabs.forEach(x=>{
      x.setAttribute('aria-selected','false');
      x.setAttribute('aria-expanded','false');
    });
    if(tabs[0]){
      tabs[0].setAttribute('aria-selected','true');
      tabs[0].setAttribute('aria-expanded','true');
    }
  }

  tabs.forEach(btn=>{
    btn.addEventListener('click', ()=>{
      if(isMobile()){
        const expanded = btn.getAttribute('aria-expanded') === 'true';
        if(expanded){
          btn.setAttribute('aria-expanded','false');
          return;
        }
        tabs.forEach(x=>x.setAttribute('aria-expanded','false'));
        btn.setAttribute('aria-expanded','true');

        tabs.forEach(x=>x.setAttribute('aria-selected','false'));
        btn.setAttribute('aria-selected','true');
      }else{
        setActive(btn);
      }
    });
  });

  window.addEventListener('resize', ()=>{
    const active = tabs.find(x => x.getAttribute('aria-selected') === 'true') || tabs[0];
    if(active && !isMobile()){
      renderDesktop(active.dataset.key || 'c1');
    }
  });
})();
</script>
