<script>
  document.addEventListener('DOMContentLoaded', function(){
    const root = document.querySelector('#offerings.co2');
    if(!root) return;
    const items  = root.querySelectorAll('.co2__item');
    const panels = root.querySelectorAll('.co2-panel');
    function activate(btn){
      const targetId = btn.getAttribute('data-target');
      items.forEach(b=>{
        const on = (b === btn);
        b.classList.toggle('is-active', on);
        b.setAttribute('aria-selected', on ? 'true' : 'false');
      });
      panels.forEach(p=>{
        p.classList.toggle('is-active', p.id === targetId);
      });
      // Update the primary CTA inside active panel
      const activePanel = root.querySelector('.co2-panel.is-active');
      const cta = activePanel ? activePanel.querySelector('.co2__btn--primary') : null;
      if(cta){
        const newText = btn.getAttribute('data-cta-text') || 'Explore';
        const newHref = btn.getAttribute('data-cta-link') || '#';
        const span = cta.querySelector('span');
        if(span) span.innerHTML = newText;
        cta.setAttribute('href', newHref);
      }
    }
    items.forEach(btn=>{
      btn.addEventListener('click', ()=>activate(btn));
    });
  });
</script>
