<script>
  (function(){
    function initAboutReveal(scope){
      // scope = elementor widget/container, or document
      const root = scope && scope.querySelector ? scope : document;
      const section = root.querySelector('#purpose-vision');
      if(!section) return;

      // prevent double-init
      if(section.dataset.abInit === "1") return;
      section.dataset.abInit = "1";

      const items = section.querySelectorAll('.ab-reveal');
      items.forEach((el,i)=>{
        setTimeout(()=> el.classList.add('is-in'), i*120);
      });
    }

    // Works on normal pages
    document.addEventListener('DOMContentLoaded', function(){
      initAboutReveal(document);
    });

    // Works inside Elementor editor + frontend renders
    if(window.elementorFrontend && window.elementorFrontend.hooks){
      window.elementorFrontend.hooks.addAction('frontend/element_ready/global', function($scope){
        // $scope is a jQuery object in Elementor; convert safely
        const el = $scope && $scope[0] ? $scope[0] : document;
        initAboutReveal(el);
      });
    }
  })();
</script>

<script>
  (function(){
    const data = {
      p1:{
        num:"01",
        title:"Client-First Mindset",
        desc:"We put your priorities at the center of everything we do, aligning strategies with your vision.",
        impact:"Measurable Impact: 95%+ client satisfaction scores, repeat engagements with over 80% of our clients, and measurable business KPI improvements within the first 3–6 months."
      },
      p2:{
        num:"02",
        title:"Employee Engagement & Ownership",
        desc:"We empower our people to take responsibility, innovate, and drive impact as if they own the outcome.",
        impact:"Measurable Impact: 90%+ project ownership scores in internal reviews, faster decision cycles, and proactive issue resolution before escalation in 95% of cases."
      },
      p3:{
        num:"03",
        title:"Ownership for Excellence",
        desc:"We hold ourselves accountable for delivering at the highest standards — every time, without compromise.",
        impact:"Measurable Impact: Zero critical defects in deliverables, >98% on-time project delivery, and client-rated quality score of 9/10 or higher."
      },
      p4:{
        num:"04",
        title:"Deliver Sustainable Results",
        desc:"We design strategies and solutions that deliver measurable, long-term value beyond the initial engagement.",
        impact:"Measurable Impact: Minimum 12-month sustainability check on implemented solutions, with over 85% of outcomes still delivering value a year later."
      },
      p5:{
        num:"05",
        title:"Ensure Positive ROI",
        desc:"We commit to delivering financial and operational returns that far exceed investment.",
        impact:"Measurable Impact: Positive ROI within 16 weeks in all engagements; target 4:1 ROI following full-scale rollout."
      }
    };

    const tabs = Array.from(document.querySelectorAll('#core-principles .cpB__item'));
    const fade = document.getElementById('cpB-fade');
    const title = document.getElementById('cpB-title');
    const badge = document.getElementById('cpB-badge');
    const desc = document.getElementById('cpB-desc');
    const impact = document.getElementById('cpB-impact');

    function setActive(btn){
      const key = btn.getAttribute('data-key');
      const d = data[key];
      if(!d) return;

      tabs.forEach(t => t.setAttribute('aria-selected', t === btn ? 'true' : 'false'));

      // subtle crossfade
      fade.classList.add('is-switching');
      setTimeout(()=>{
        badge.textContent = d.num;
        title.textContent = d.title;
        desc.textContent = d.desc;
        impact.textContent = d.impact;
        fade.classList.remove('is-switching');
      }, 160);
    }

    tabs.forEach(btn=>{
      btn.addEventListener('click', ()=> setActive(btn));
      btn.addEventListener('keydown', (e)=>{
        const i = tabs.indexOf(btn);
        if(e.key === 'ArrowDown' || e.key === 'ArrowRight'){
          e.preventDefault();
          const next = tabs[i+1] || tabs[0];
          next.focus(); setActive(next);
        }
        if(e.key === 'ArrowUp' || e.key === 'ArrowLeft'){
          e.preventDefault();
          const prev = tabs[i-1] || tabs[tabs.length-1];
          prev.focus(); setActive(prev);
        }
      });
    });
  })();
</script>
