<script>
(function(){
  const nav = document.querySelector('.gccTabs');
  if(!nav) return;

  const links = Array.from(nav.querySelectorAll('.gccTabs__link'));
  const getOffset = () => (nav.getBoundingClientRect().height || 58) + 12;

  // Smooth scroll with correct offset
  links.forEach(a=>{
    a.addEventListener('click', (e)=>{
      const href = a.getAttribute('href');
      if(!href || !href.startsWith('#')) return;

      const el = document.querySelector(href);
      if(!el) return;

      e.preventDefault();
      const y = el.getBoundingClientRect().top + window.pageYOffset - getOffset();
      window.scrollTo({ top:y, behavior:'smooth' });
      history.replaceState(null, '', href);
    });
  });

  // Active state on scroll
  const targets = links
    .map(a => document.querySelector(a.getAttribute('href')))
    .filter(Boolean);

  const setActive = (id) => {
    links.forEach(a=>{
      a.classList.toggle('is-active', a.getAttribute('href') === '#'+id);
    });
  };

  const isElementor =
    document.body.classList.contains('elementor-editor-active') ||
    document.body.classList.contains('elementor-editor-preview') ||
    !!document.querySelector('#elementor-editor-wrapper');

  if(isElementor) return;
  if(!('IntersectionObserver' in window)) return;

  const io = new IntersectionObserver((entries)=>{
    const visible = entries
      .filter(e=>e.isIntersecting)
      .sort((a,b)=>b.intersectionRatio-a.intersectionRatio)[0];

    if(visible && visible.target && visible.target.id){
      setActive(visible.target.id);
    }
  }, { root:null, threshold:[0.2,0.35,0.5,0.65] });

  targets.forEach(t=>{ if(t.id) io.observe(t); });
})();
</script>

        "Limited, unclear, or narrow career pathways impacting retention and engagement",
        "Inconsistent or fragmented capability development frameworks across roles and functions"
      ],
      chips:["Talent","Capability","Leadership"]
    },
    c4:{
      title:"Productivity, Cost & Performance Pressure",
      bullets:[
        "Rising operating costs gradually eroding traditional labour-arbitrage advantages",
        "Limited function-wise productivity benchmarks and peer comparisons",
        "Manual processes and underutilised automation across services and workflows",
        "Weak visibility into performance, service levels, and business outcomes across the GCC portfolio"
      ],
      chips:["Productivity","Cost","Performance"]
    },
    c5:{
      title:"Digital, Data & Technology Fragmentation",
      bullets:[
        "Siloed platforms and tools across IT, analytics, finance, HR, and operations",
        "Ad-hoc automation initiatives without alignment to enterprise architecture or priorities",
        "Limited use of data for performance management, demand forecasting, and capacity planning",
        "Challenges in scaling AI and advanced analytics responsibly across services and geographies"
      ],
      chips:["Digital","Data & analytics","Automation & AI"]
    }
  };

  const root = document.querySelector('#gcc-challenges');
  if(!root) return;

  function applyMobileClass(){
    const mqMobile = window.matchMedia && window.matchMedia('(max-width: 820px)').matches;

    // Elementor responsive preview helpers
    const elMobile =
      document.body.classList.contains('elementor-device-mobile') ||
      document.body.classList.contains('e--ua-device-mobile');

    const mobile = mqMobile || elMobile;

    root.classList.toggle('is-mobile', mobile);
    return mobile;
  }
  applyMobileClass();
  window.addEventListener('resize', applyMobileClass);

  const tabs = Array.from(root.querySelectorAll('.gccK__item'));
  const fade = root.querySelector('#gccK-fade');
  const title = root.querySelector('#gccK-title');
  const bulletsEl = root.querySelector('#gccK-bullets');
  const chipsEl = root.querySelector('#gccK-chips');

  function isMobile(){
    return (window.matchMedia && window.matchMedia('(max-width: 820px)').matches) || root.classList.contains('is-mobile');
  }

  function renderDesktop(key){
    const d = data[key];
    if(!d || !fade) return;

    fade.classList.add('is-switching');
    setTimeout(()=>{
      title.textContent = d.title;

      bulletsEl.innerHTML = d.bullets.map(x =>
        `<li><span class="gccK__dot"></span><span>${x}</span></li>`
      ).join('');

      chipsEl.innerHTML = d.chips.map(x =>
        `<span class="gccK__chip"><i></i>${x}</span>`
      ).join('');

      fade.classList.remove('is-switching');
    }, 160);
  }

  function renderAccordionItem(btn){
    const key = btn.dataset.key;
    const d = data[key];
    if(!d) return;

    const acc = btn.querySelector('.gccK__acc');
    if(!acc) return;

    acc.innerHTML = `
      <ul class="gccK__accBullets">
        ${d.bullets.map(x => `<li><span class="gccK__dot"></span><span>${x}</span></li>`).join('')}
      </ul>
      <div class="gccK__accChips">
        ${d.chips.map(x => `<span class="gccK__chip"><i></i>${x}</span>`).join('')}
      </div>
    `;
  }

  tabs.forEach(btn => renderAccordionItem(btn));

  function setActive(btn){
    tabs.forEach(x=>{
      x.setAttribute('aria-selected','false');
      x.setAttribute('aria-expanded','false');
    });
    btn.setAttribute('aria-selected','true');

    if(isMobile()){
      btn.setAttribute('aria-expanded','true');
    }else{
      btn.setAttribute('aria-expanded','false');
      renderDesktop(btn.dataset.key);
    }
  }

  // Desktop default
  renderDesktop('c1');

  // Mobile: open first card by default
  if(isMobile()){
    tabs.forEach(x=>{
      x.setAttribute('aria-selected','false');
      x.setAttribute('aria-expanded','false');
    });
    if(tabs[0]){
      tabs[0].setAttribute('aria-selected','true');
      tabs[0].setAttribute('aria-expanded','true');
    }
  }

  tabs.forEach(btn=>{
    btn.addEventListener('click', ()=>{
      if(isMobile()){
        const expanded = btn.getAttribute('aria-expanded') === 'true';
        if(expanded){
          btn.setAttribute('aria-expanded','false');
          return;
        }
        tabs.forEach(x=>x.setAttribute('aria-expanded','false'));
        btn.setAttribute('aria-expanded','true');

        tabs.forEach(x=>x.setAttribute('aria-selected','false'));
        btn.setAttribute('aria-selected','true');
      }else{
        setActive(btn);
      }
    });
  });

  window.addEventListener('resize', ()=>{
    const active = tabs.find(x => x.getAttribute('aria-selected') === 'true') || tabs[0];
    if(active && !isMobile()){
      renderDesktop(active.dataset.key || 'c1');
    }
  });
})();
