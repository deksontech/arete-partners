<script>
(function(){
  const nav = document.querySelector('.secTabs');
  if(!nav) return;

  const links = Array.from(nav.querySelectorAll('.secTabs__link'));
  const targets = links
    .map(a => {
      const href = a.getAttribute('href');
      return (href && href.startsWith('#')) ? document.querySelector(href) : null;
    })
    .filter(Boolean);

  // Smooth scroll
  links.forEach(a=>{
    a.addEventListener('click', (e)=>{
      const id = a.getAttribute('href');
      if(!id || !id.startsWith('#')) return;
      const el = document.querySelector(id);
      if(!el) return;
      e.preventDefault();

      // If you ever add a sticky main header later, set this to its height (e.g. 80)
      const headerOffset = 0;

      const y = el.getBoundingClientRect().top + window.pageYOffset - headerOffset;
      window.scrollTo({ top:y, behavior:'smooth' });
      history.replaceState(null, '', id);
    });
  });

  const setActive = (id) => {
    links.forEach(a=>{
      a.classList.toggle('is-active', a.getAttribute('href') === '#'+id);
    });
  };

  // Elementor editor: keep first active
  const isElementor =
    document.body.classList.contains('elementor-editor-active') ||
    document.body.classList.contains('elementor-editor-preview') ||
    !!document.querySelector('#elementor-editor-wrapper');

  if(isElementor) return;

  if(!('IntersectionObserver' in window)) return;

  const io = new IntersectionObserver((entries)=>{
    const visible = entries
      .filter(e=>e.isIntersecting)
      .sort((a,b)=>b.intersectionRatio-a.intersectionRatio)[0];

    if(visible && visible.target && visible.target.id){
      setActive(visible.target.id);
    }
  }, { threshold:[0.2,0.35,0.5,0.65] });

  targets.forEach(t=>{
    if(t.id) io.observe(t);
  });
})();
</script>

<script>
  (function(){
    const root = document.getElementById('im-how-we-help');
    if(!root) return;

    const track = root.querySelector('#imh-track');
    const slides = Array.from(root.querySelectorAll('.imhSlide'));
    const prevBtn = root.querySelector('#imh-prev');
    const nextBtn = root.querySelector('#imh-next');

    let index = 0;

    function clamp(n,min,max){ return Math.max(min, Math.min(max,n)); }

    function update(){
      // slide width includes gap; compute from first slide
      const slide = slides[0];
      const slideW = slide.getBoundingClientRect().width;
      const gap = parseFloat(getComputedStyle(track).gap || "16");

      // Move track so active slide is centered-ish with peeks visible
      const offset = (slideW + gap) * index;
      track.style.transform = `translateX(${-offset}px)`;

      prevBtn.disabled = (index === 0);
      nextBtn.disabled = (index === slides.length - 1);
    }

    prevBtn.addEventListener('click', ()=>{
      index = clamp(index - 1, 0, slides.length - 1);
      update();
    });

    nextBtn.addEventListener('click', ()=>{
      index = clamp(index + 1, 0, slides.length - 1);
      update();
    });

    // touch / drag
    let startX = null;
    let startT = 0;

    function onDown(e){
      startX = (e.touches ? e.touches[0].clientX : e.clientX);
      const m = getComputedStyle(track).transform;
      startT = m && m !== "none" ? new DOMMatrixReadOnly(m).m41 : 0;
      track.style.transition = "none";
    }
    function onMove(e){
      if(startX === null) return;
      const x = (e.touches ? e.touches[0].clientX : e.clientX);
      const dx = x - startX;
      track.style.transform = `translateX(${startT + dx}px)`;
    }
    function onUp(e){
      if(startX === null) return;

      track.style.transition = "transform .35s ease";

      const slideW = slides[0].getBoundingClientRect().width;
      const gap = parseFloat(getComputedStyle(track).gap || "16");
      const step = slideW + gap;

      const m = getComputedStyle(track).transform;
      const cur = m && m !== "none" ? new DOMMatrixReadOnly(m).m41 : 0;

      index = clamp(Math.round(Math.abs(cur) / step), 0, slides.length - 1);

      startX = null;
      update();
    }

    track.addEventListener('mousedown', onDown);
    window.addEventListener('mousemove', onMove);
    window.addEventListener('mouseup', onUp);

    track.addEventListener('touchstart', onDown, {passive:true});
    track.addEventListener('touchmove', onMove, {passive:true});
    track.addEventListener('touchend', onUp);

    // reveal left/right when section enters viewport (Elementor safe)
    const isElementorEditor =
      document.body.classList.contains('elementor-editor-active') ||
      document.body.classList.contains('elementor-editor-preview') ||
      !!document.querySelector('#elementor-editor-wrapper');

    function show(){ root.classList.add('is-in'); }

    if(isElementorEditor || !('IntersectionObserver' in window)){
      show();
    }else{
      const io = new IntersectionObserver(entries=>{
        entries.forEach(ent=>{
          if(ent.isIntersecting){
            show();
            io.disconnect();
          }
        });
      }, {threshold:0.18});
      io.observe(root);
      setTimeout(()=>{ if(!root.classList.contains('is-in')) show(); }, 1200);
    }

    // init
    window.addEventListener('load', update);
    window.addEventListener('resize', ()=>{ track.style.transition="none"; update(); setTimeout(()=>track.style.transition="transform .35s ease", 0); });
  })();
</script>
