  const btnNext = section.querySelector('.phhS__ctrl--next');

  let idx = 0;

  function cardHTML(d){
    return `
      <div class="phhS__top">
        <p class="phhS__kicker">${d.kicker}</p>
        <h3 class="phhS__cardTitle">${d.title}</h3>
      </div>

        <p class="phhS__lead">${d.lead}</p>
        <div class="phhS__label">${d.label1}</div>
        <ul class="phhS__list">
          ${d.bullets.map(x=>`<li>${x}</li>`).join('')}
        </ul>
        <div class="phhS__impact"><b>Impact:</b> ${d.impact}</div>
      </div>
    `;
  }

  function setActiveNav(i){
    navButtons.forEach((b,bi)=> b.setAttribute('aria-selected', bi===i ? 'true':'false'));
  }

  function render(){
    const prevI = (idx - 1 + data.length) % data.length;
    const nextI = (idx + 1) % data.length;

    // Render cards
    prevEl.innerHTML = cardHTML(data[prevI]);
    activeEl.innerHTML = cardHTML(data[idx]);
    nextEl.innerHTML = cardHTML(data[nextI]);

    setActiveNav(idx);
  }

  function goTo(i){
    idx = (i + data.length) % data.length;
    render();
  }

  navButtons.forEach(btn=>{
    btn.addEventListener('click', ()=>{
      const i = parseInt(btn.getAttribute('data-i'),10);
      goTo(i);
    });
  });

  btnPrev && btnPrev.addEventListener('click', ()=> goTo(idx-1));
  btnNext && btnNext.addEventListener('click', ()=> goTo(idx+1));

  // keyboard support
  section.addEventListener('keydown', (e)=>{
    if(e.key === 'ArrowLeft') goTo(idx-1);
    if(e.key === 'ArrowRight') goTo(idx+1);
  });

  render();
})();
</script>
